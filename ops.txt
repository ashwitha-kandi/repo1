int isKeyword(char *str) {
for (int i = 0; i < 10; i++)
if (strcmp(str, keywords[i]) == 0)
return 1;
return 0;

}

int isOperator(char ch) {
return strchr("+-*/=<>%", ch) != NULL;
}

int main() {
char ch, str[50];
int i = 0;

while ((ch = getchar()) != EOF) {
// Identifier or Keyword
if (isalpha(ch) || ch == '_') {
str[i++] = ch;
while (isalnum(ch = getchar()) || ch == '_')
str[i++] = ch;
str[i] = '\0'; i = 0;
ungetc(ch, stdin);
printf("%s: %s\n", isKeyword(str) ? "Keyword" : "Identifier", str);
}

// Number (including float)
else if (isdigit(ch)) {
str[i++] = ch;
int dot_seen = 0;
while (isdigit(ch = getchar()) || (ch == '.' && !dot_seen)) {
if (ch == '.') dot_seen = 1;
str[i++] = ch;
}
str[i] = '\0'; i = 0;
ungetc(ch, stdin);

printf("Constant: %s\n", str);
}

// Comments and division operator
else if (ch == '/') {
char next = getchar();
if (next == '/') {
while ((ch = getchar()) != '\n' && ch != EOF);
printf("Comment: Single-line\n");
} else if (next == '*') {
while (1) {
ch = getchar();
if (ch == '*' && getchar() == '/') break;
}
printf("Comment: Multi-line\n");
} else {
ungetc(next, stdin);
printf("Operator: /\n");
}
}

// Operators
else if (isOperator(ch)) {
printf("Operator: %c\n", ch);
}

// Skip whitespace and punctuation
else if (isspace(ch) || ch == ';' || ch == '(' || ch == ')' ||
ch == '{' || ch == '}' || ch == ',' || ch == '.') {
continue;
}

}

return 0;

} Var;

int top = -1;
// Function to push a variable onto the stack
void push(char name[], int value) {
if (top < SIZE - 1) {
top++;
strcpy(stack[top].name, name);
stack[top].value = value;
printf("Allocated %s = %d\n", name, value);
} else {
printf("Stack Overflow: Cannot allocate %s\n", name);
}
}
// Function to pop a variable from the stack

void pop() {
if (top >= 0) {
printf("Deallocated %s\n", stack[top].name);
top--;
} else {
printf("Stack Underflow: No variable to deallocate\n");
}
}

void display() {
printf("Stack:\n");
for (int i = top; i >= 0; i--)
printf(" %s = %d\n", stack[i].name, stack[i].value);
}
